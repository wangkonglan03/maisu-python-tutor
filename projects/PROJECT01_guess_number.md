## 🎯 游戏规则

1. 电脑想一个 1-100 的数字
2. 你有 5 次猜测机会
3. 每次猜测后，电脑告诉你"太大"或"太小"
4. 猜对了就赢，超次数就输

---

## 📝 代码

```python
import random

def 猜数字游戏():
    # 电脑想数字
    答案 = random.randint(1, 100)
    猜测次数 = 0
    最大次数 = 5
    
    print("=" * 30)
    print("🎮 猜数字游戏")
    print("=" * 30)
    print("我想了一个1-100的数字")
    print(f"你有{最大次数}次机会，猜猜是多少？")
    print()
    
    while 猜测次数 < 最大次数:
        剩余次数 = 最大次数 - 猜测次数
        
        # 获取玩家输入
        猜测 = input(f"还有{剩余次数}次机会，请输入你的猜测：")
        
        # 验证输入是否是数字
        if not 猜测.isdigit():
            print("⚠️ 请输入一个数字！\n")
            continue
        
        猜测 = int(猜测)
        
        if 猜测 < 1 or 猜测 > 100:
            print("⚠️ 请输入1-100之间的数字！\n")
            continue
        
        猜测次数 = 猜测次数 + 1
        
        # 判断
        if 猜测 == 答案:
            print(f"🎉 恭喜你！猜对了！用了{猜测次数}次")
            print("=" * 30)
            return True
        elif 猜测 < 答案:
            print("📈 太小了，往大猜！")
        else:
            print("📉 太大了，往小猜！")
        print()
    
    # 次数用完
    print(f"😢 次数用完了！正确答案是{答案}")
    print("=" * 30)
    return False

# 运行游戏
猜数字游戏()
```
